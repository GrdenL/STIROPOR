SQL cleanup commands (Postgres)

1) List all tables (verify actual names):
SELECT tablename
FROM pg_tables
WHERE schemaname = 'public'
ORDER BY tablename;

2) Truncate everything except games + genres (auto):
DO $$
DECLARE r record;
BEGIN
  FOR r IN
    SELECT tablename
    FROM pg_tables
    WHERE schemaname = 'public'
      AND tablename NOT IN ('games', 'genres')
  LOOP
    EXECUTE format('TRUNCATE TABLE %I RESTART IDENTITY CASCADE;', r.tablename);
  END LOOP;
END $$;
